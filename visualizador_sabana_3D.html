    <script>

        window.addEventListener('load', async function() {

            const initialMessageDiv = document.getElementById('initialMessage');

            const controlsPanel = document.getElementById('controlsPanel');

            

            try {

                console.log("Página cargada. Iniciando Cesium...");

                console.log("CESIUM_BASE_URL está configurada como:", window.CESIUM_BASE_URL);



                // Token de Acceso de Cesium Ion. 

                Cesium.Ion.defaultAccessToken = 'eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJqdGkiOiIwMTZiNWVjNi1jZTI3LTRjNjktOTUwMi04NWZlYTdiM2M4ZDIiLCJpZCI6MzAyNTUwLCJpYXQiOjE3NDcyNjA3MDV9.a54xRw4uqeucut0ipo8dcYrycXNHMkq4YtoXThS4taU'; 


                if (Cesium.Ion.defaultAccessToken === 'TU_CESIUM_ION_ACCESS_TOKEN') {

                    throw new Error("INVALID_TOKEN_PLACEHOLDER");

                }


                const initialLongitude = -74.08175;

                const initialLatitude = 4.60971;

                const initialHeight = 30000; 


                const chkWireframe = document.getElementById('chkWireframe');

                const chkContours = document.getElementById('chkContours');

                const btnResetView = document.getElementById('btnResetView');

                

                let originalGlobeMaterial;

                let viewer;


                if (typeof Cesium === 'undefined') throw new Error("Biblioteca Cesium no definida. Verifica la conexión a internet y la URL del script.");

                if (typeof Cesium.createWorldTerrainAsync !== 'function') throw new Error("Cesium.createWorldTerrainAsync no es una función. La biblioteca Cesium puede estar desactualizada o incompleta.");


                let terrainProvider;

                try {

                    console.log("Intentando cargar proveedor de terreno mundial de Cesium Ion...");

                    terrainProvider = await Cesium.createWorldTerrainAsync();

                    console.log("Proveedor de terreno de Cesium Ion cargado exitosamente.");

                } catch (terrainError) {

                    console.error("Error al cargar el proveedor de terreno de Cesium Ion:", terrainError);

                    if (initialMessageDiv.innerHTML.includes("Iniciando")) initialMessageDiv.innerHTML = ""; // Limpiar mensaje inicial

                    initialMessageDiv.innerHTML += "<br><b>Advertencia:</b> No se pudo cargar el terreno mundial de Cesium Ion. Se usará un globo sin relieve detallado. Error: " + (terrainError.message || terrainError.toString());

                    initialMessageDiv.style.display = 'block';

                }

                

                let imageryProvider;

                try {

                    console.log("Intentando cargar proveedor de imágenes mundial de Cesium Ion...");

                    if (typeof Cesium.createWorldImageryAsync === 'function') {

                        imageryProvider = await Cesium.createWorldImageryAsync();

                        console.log("Proveedor de imágenes de Cesium Ion cargado exitosamente.");

                    } else {

                         console.warn("Cesium.createWorldImageryAsync no disponible. Se intentará un proveedor de respaldo.");

                         throw new Error("createWorldImageryAsync not available"); 

                    }

                } catch (imageryError) {

                    console.error("Error al cargar el proveedor de imágenes de Cesium Ion:", imageryError);

                    console.log("Intentando cargar OpenStreetMap como proveedor de imágenes de respaldo...");

                    try {

                        imageryProvider = new Cesium.OpenStreetMapImageryProvider({

                            url : 'https://a.tile.openstreetmap.org/'

                        });

                        console.log("Proveedor de imágenes OpenStreetMap cargado exitosamente como respaldo.");

                        if (initialMessageDiv.innerHTML.includes("Iniciando")) initialMessageDiv.innerHTML = "";

                        initialMessageDiv.innerHTML += "<br><b>Info:</b> Usando imágenes de OpenStreetMap como respaldo.";

                        initialMessageDiv.style.display = 'block';

                    } catch (osmError) {

                        console.error("Error al cargar OpenStreetMap como proveedor de imágenes de respaldo:", osmError);

                        if (initialMessageDiv.innerHTML.includes("Iniciando")) initialMessageDiv.innerHTML = "";

                        initialMessageDiv.innerHTML += "<br><b>Advertencia:</b> No se pudo cargar imágenes de Cesium Ion ni de OpenStreetMap. El globo podría aparecer negro o solo con estrellas. Error: " + (osmError.message || osmError.toString());

                        initialMessageDiv.style.display = 'block';

                    }

                }


                viewer = new Cesium.Viewer('cesiumContainer', {

                    terrainProvider: terrainProvider, 

                    imageryProvider: imageryProvider, 

                    animation: false, baseLayerPicker: false, fullscreenButton: true,

                    geocoder: false, homeButton: false, infoBox: true,

                    sceneModePicker: true, selectionIndicator: true, timeline: false,

                    navigationHelpButton: false,

                });

                console.log("Cesium Viewer inicializado.");


                if (viewer.scene && viewer.scene.globe) {

                    viewer.scene.globe.show = true;

                    console.log("Globo configurado para ser visible (viewer.scene.globe.show = true).");

                } else {

                     console.error("viewer.scene.globe no está disponible después de la inicialización del visor.");

                }


                // Ocultar mensaje inicial solo si no hay advertencias/errores importantes mostrados

                if (initialMessageDiv.style.display !== 'block' || initialMessageDiv.innerHTML.includes("Iniciando Visualizador Cesium...")) {

                    initialMessageDiv.style.display = 'none';

                }

                if (controlsPanel) controlsPanel.style.display = 'block';


                if (viewer.scene && viewer.scene.globe) originalGlobeMaterial = viewer.scene.globe.material;

                else console.error("viewer.scene.globe no disponible para guardar material original.");


                viewer.camera.flyTo({ 

                    destination: Cesium.Cartesian3.fromDegrees(initialLongitude, initialLatitude, initialHeight),

                    orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-45.0), roll: 0.0 },

                    duration: 0 

                });

                console.log("Vista de cámara configurada.");


                chkWireframe.addEventListener('change', function() { if (viewer.scene.globe) viewer.scene.globe.wireframe = this.checked; });

                chkContours.addEventListener('change', function() {

                    if (viewer.scene.globe) {

                        if (this.checked) {

                            try {

                                viewer.scene.globe.material = Cesium.Material.fromType('ElevationContour', {

                                    spacing: 100.0, contourColor: Cesium.Color.RED.withAlpha(0.7), contourWidth: 1.5,

                                });

                            } catch (e) { console.error("Error al aplicar material de curvas de nivel:", e); }

                        } else {

                            if (originalGlobeMaterial !== undefined) viewer.scene.globe.material = originalGlobeMaterial;

                        }

                    }

                });

                btnResetView.addEventListener('click', function() {

                    if (viewer.camera) viewer.camera.flyTo({

                        destination: Cesium.Cartesian3.fromDegrees(initialLongitude, initialLatitude, initialHeight),

                        orientation: { heading: Cesium.Math.toRadians(0.0), pitch: Cesium.Math.toRadians(-45.0), roll: 0.0 },

                        duration: 2.0

                    });

                });


            } catch (error) {

                console.error("Error CRÍTICO al inicializar Cesium:", error);

                let userFriendlyMessage = "<b>Error al cargar el visor 3D:</b><br>";

                let errorDetails = "";


                if (error && error.message && error.message === "INVALID_TOKEN_PLACEHOLDER") {

                    // ... (código de manejo de error de token de placeholder)

                } else if (error && error.message) {

                    errorDetails = error.message;

                     if (error.message.toLowerCase().includes("invalid access token") || (error.statusCode === 401)) {

                        errorDetails += "<br><br><b>Sugerencia:</b> El token de Cesium Ion que proporcionaste parece ser inválido o ha expirado. Verifica tu token en <a href='https://cesium.com/ion/tokens' target='_blank'>cesium.com/ion/tokens</a> y actualízalo en el código.";

                    }

                } else if (error && error.response) { 

                    // ... (código de manejo de error de respuesta de Cesium)

                } else if (typeof error === 'string') {

                    errorDetails = error;

                } else {

                    errorDetails = "Ocurrió un error desconocido durante la inicialización: " + (error ? error.toString() : "Error N/A");

                }

                

                userFriendlyMessage += errorDetails;

                userFriendlyMessage += "<br><br>Revisa la consola del desarrollador (F12) para más detalles técnicos. Asegúrate de tener conexión a internet.";


                if (initialMessageDiv) {

                    initialMessageDiv.innerHTML = userFriendlyMessage;

                    initialMessageDiv.style.display = 'block';

                }

                if (controlsPanel) controlsPanel.style.display = 'none';

            }

        });

    </script>
